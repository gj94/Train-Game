async function ranges(relative,local){
 const file=await open(local,'r'),size=(await file.stat()).size;
 const head=await fetch(base+relative,{method:'HEAD'});assert.equal(head.status,200);assert.equal(Number(head.headers.get('content-length')),size);
 for(const start of [0,Math.floor(size/2),Math.max(0,size-65536)]){
  const end=Math.min(size-1,start+65535),response=await fetch(base+relative,{headers:{Range:`bytes=${start}-${end}`}});
  assert.equal(response.status,206);assert.equal(response.headers.get('content-range'),`bytes ${start}-${end}/${size}`);
  const expected=Buffer.alloc(end-start+1);await file.read(expected,0,expected.length,start);assert.deepEqual(Buffer.from(await response.arrayBuffer()),expected);
 }
 await file.close();return size;
}

 if([0,Math.floor(core.chunks.length/2),core.chunks.length-1].includes(i)){
