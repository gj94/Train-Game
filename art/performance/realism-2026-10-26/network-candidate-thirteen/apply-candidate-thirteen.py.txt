from pathlib import Path
import hashlib, json, shutil, subprocess

root = Path('D:/ClaudeWS/train-game')
local = root / '.local/realism-2026-10-26'
archive = local / 'candidate-thirteen-before'
assert not archive.exists(), 'Application already started; inspect instead of repeating'
shell = 'C:/Users/Gokul/.cache/codex-runtimes/codex-primary-runtime/dependencies/native/powershell/pwsh.exe'
active = subprocess.check_output([shell, '-NoProfile', '-Command',
    "@(Get-CimInstance Win32_Process | Where-Object { $_.Name -like 'Godot*' -and ($_.CommandLine -like '*--script res://tools/prepare_timetable_variant.gd*' -or $_.CommandLine -like '*--script res://tests/run_tests.gd*') }).Count"], text=True).strip()
assert active == '0', active
paths = [root / 'art/timetables' / n for n in ['kerala-coast-network.json', 'kerala-coast-network-v2.json', 'kerala-coast-network-departure-overrides.json']]
packs = [json.loads(p.read_text(encoding='utf-8')) for p in paths[:2]]
assert packs[0]['services'] == packs[1]['services']
services = {s['id']: s for s in packs[0]['services']}
assert services['L002']['departure'] == '05:05'
assert services['M064']['departure'] == '04:10'
assert services['L034']['departure'] == '04:30'
assert services['K1']['departure'] == '06:00'
overrides = json.loads(paths[2].read_text(encoding='utf-8'))
rows = [r for r in overrides['changes'] if r['service'] == 'L002']
assert len(rows) == 1
rows[0]['to'] = '04:20'
rows[0]['reason'] = 'Network candidate thirteen: place L002 after M06404:10 and before L03404:30, avoiding the shared05:05 origin with S030 and the observed later NEM opposing wave. Departure-only hypothesis; complete baseline and disturbance acceptance remains required.'
archive.mkdir()
for p in paths:
    shutil.copyfile(p, archive / p.name)
    committed = subprocess.check_output(['git','-c','safe.directory=D:/ClaudeWS/train-game','show','HEAD:'+p.relative_to(root).as_posix()])
    assert json.loads(committed) == json.loads(p.read_bytes())
    if p.name != 'kerala-coast-network-departure-overrides.json':
        assert committed == p.read_bytes()
paths[2].write_text(json.dumps(overrides,ensure_ascii=False,indent=2)+'\n',encoding='utf-8')
(local/'candidate-thirteen-plan.json').write_text(json.dumps(dict(
    predecessors={p.name:hashlib.sha256((archive/p.name).read_bytes()).hexdigest() for p in paths},
    changes=[dict(service='L002',before='05:05',after='04:20')],
    evidence_commit='6407d5a9',status='Departure-only hypothesis; no traffic acceptance'
),ensure_ascii=False,indent=2)+'\n',encoding='utf-8')
print('Updated one override; packs await normal authoring')
