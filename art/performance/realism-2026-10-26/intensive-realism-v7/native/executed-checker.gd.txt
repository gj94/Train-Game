extends SceneTree
## Use only after certified baselines are installed. Real menu loaders and
## timeline workers, isolated preferences/saves, no alternate fixture catalogues.
const Days:=preload("res://persistence/operating_days.gd")
const Traffic:=preload("res://persistence/service_traffic.gd")
var OUT:="res://.local/dispatcher-v2-2026-10-24/prepared-ui"
var checks:=0
var failures:=[]
var results:=[]
var requested:=[]
func _initialize() -> void:call_deferred("run")
func check(ok: bool,label: String) -> bool:
	checks+=1
	if not ok:failures.append(label);printerr("FAIL ",label)
	return ok
func frames() -> void:
	for i in 4:await process_frame
func shot(label: String) -> void:
	await frames()
	if DisplayServer.get_name()=="headless":return
	await RenderingServer.frame_post_draw
	root.get_texture().get_image().save_png(OUT+"/"+label+".png")
func run() -> void:
	for arg in OS.get_cmdline_user_args():
		if arg.begins_with("--day="):requested.append(arg.trim_prefix("--day="))
		if arg.begins_with("--output="):OUT=arg.trim_prefix("--output=")
	for id: String in requested:
		if not Days.DAYS.any(func(day):return day.id==id):printerr("Unknown day ",id);quit(1);return
	DirAccess.make_dir_recursive_absolute(ProjectSettings.globalize_path(OUT))
	root.size=Vector2i(1280,720);root.content_scale_size=root.size
	for index in Days.DAYS.size():
		var day: Dictionary=Days.DAYS[index]
		if not requested.is_empty() and day.id not in requested:continue
		if not check(Days.available(day.id),day.id+" certified catalogues available"):continue
		var page:=preload("res://game/launch.gd").new()
		page.browser_preferences=OUT+"/browser.cfg"
		page.store=preload("res://persistence/save_store.gd").new(OUT+"/saves")
		root.add_child(page);await frames();page.show_browser();page.select_timetable(index)
		page.reset_filters();page.selected={id="K1"};page.refresh();await frames()
		check(page.services.size()==day.count,day.id+" service count")
		check(page.selected.get("id","")=="K1",day.id+" passenger selectable")
		check(page.directory==day.starts,day.id+" isolated departure directory")
		check(page.calls_grid.get_child_count()==(page.selected.get("calls",[]).size()+1)*3,day.id+" all booked calls shown")
		page.set_detail_mode(1);await frames()
		check(page.traffic_summary.text.begins_with("Predictive V2 AI baseline" if day.dispatcher=="predictive" else "Classic AI baseline"),day.id+" traffic mode label")
		var overtakers:={}
		for row: Dictionary in page.traffic_rows:
			if row.role=="Overtakes you":overtakers[row.other]=true
		if day.count==140:check(overtakers.size()>=10,day.id+" ten distinct measured K1 overtakers")
		check(not page.traffic_rows.is_empty(),day.id+" measured traffic available")
		await shot(day.id+"-traffic")
		var entry:=page._read_request("service",page.selected)
		if check(entry.ok,day.id+" departure restoration"):
			check(entry.timetable_id==day.id and entry.world.trains.size()==day.count,day.id+" restored railway identity")
			check(entry.world.dispatcher().predictive.enabled==(day.dispatcher=="predictive"),day.id+" restored dispatch mode")
		# An actual counterpart list must show the reverse role for every K1 pass.
		for other: String in overtakers:
			var reciprocal:=Traffic.for_service(page.traffic_data,other)
			check(reciprocal.any(func(row):return row.other=="K1" and row.role=="You overtake"),day.id+" reciprocal pass "+other)
		if day.count==208:await network_browser(page,day)
		page.queue_free();await frames()
		set_meta("timetable_id",day.id)
		var timeline:=preload("res://game/launch_timeline.gd").new();root.add_child(timeline);await frames()
		var noon:=timeline.states.find_custom(func(row):return row.clock==43200)
		if check(noon>=0,day.id+" noon checkpoint available"):
			timeline.choose(noon)
			var started:=Time.get_ticks_msec()
			while (timeline.loaded_index!=noon or timeline.worker!=null) and Time.get_ticks_msec()-started<60000:await process_frame
			if check(timeline.world!=null and timeline.loaded_index==noon,day.id+" real timeline worker restoration"):
				check(timeline.world.trains.size()==day.count and is_equal_approx(timeline.world.clock_seconds(),43200),day.id+" noon railway identity")
				check(timeline.world.dispatcher().predictive.enabled==(day.dispatcher=="predictive"),day.id+" noon dispatch mode")
				check(timeline.services.any(func(t):return t.id=="K1"),day.id+" K1 available at noon")
				timeline._select_train("K1");await shot(day.id+"-noon")
		if day.count==208:await network_midnight(timeline,day)
		results.append({day=day.id,k1_distinct_overtakers=overtakers.size(),timeline_loaded=timeline.world!=null})
		timeline.queue_free();await frames()
	remove_meta("timetable_id")
	var report:={ok=failures.is_empty(),checks=checks,failures=failures,days=results}
	FileAccess.open(OUT+"/result.json",FileAccess.WRITE).store_string(JSON.stringify(report,"\t"))
	print("PREPARED_UI ",JSON.stringify(report));quit(0 if report.ok else 1)

func network_browser(page,day: Dictionary) -> void:
	page.reset_filters();page.selected={id="M061"};page.refresh();await frames()
	if not check(page.selected.get("id","")=="M061",day.id+" through mail selectable"):return
	var journey: Dictionary=page.selected.get("journey",{})
	if not check(journey.has("origin") and journey.has("destination"),day.id+" full journey metadata available"):return
	var index: int=page.shown.find_custom(func(row):return row.id=="M061")
	var text: String=page.roster.get_item_text(index)
	check(text.contains(journey.origin) and text.contains(journey.destination),day.id+" roster shows off-map journey endpoints")
	check(page.departure.text.ends_with(" corridor entry"),day.id+" portal time labelled as corridor entry")
	check(page.description.text.contains("Representative game service"),day.id+" authored adaptation labelled honestly")
	page.library.through_only=true;page.refresh();page.set_detail_mode(0);await frames()
	check(page.shown.any(func(row):return row.id=="M061"),day.id+" Full route retains boundary through working")
	check(page.calls_grid.get_child_count()==(page.selected.calls.size()+1)*3,day.id+" through calls and boundaries displayed")
	var entry: Dictionary=page._read_request("service",page.selected)
	if check(entry.ok,day.id+" actual portal departure restoration"):
		check(entry.world.scenery.get("boundary_network",false) and entry.world.trains.M061.journey==journey,day.id+" restored through identity and boundary railway")
	page._save_browser()
	var remembered:=preload("res://game/journey_library.gd").new();remembered.restore(page.browser_preferences)
	check(remembered.timetable==day.id and remembered.selections.get(day.id,"")=="M061",day.id+" actual per-day through selection persists")
	await shot(day.id+"-through")
	page.reset_filters();page.set_period(4);await frames()
	check(not page.shown.is_empty() and page.shown.all(func(row):return page.in_period(row.departure,4)),day.id+" overnight services selectable in Night filter")

func network_midnight(timeline,day: Dictionary) -> void:
	var midnight: int=timeline.states.find_custom(func(row):return absf(row.clock-86400)<.001)
	if not check(midnight>=0,day.id+" next-day midnight checkpoint available"):return
	timeline.choose(midnight)
	var started:=Time.get_ticks_msec()
	while (timeline.loaded_index!=midnight or timeline.worker!=null) and Time.get_ticks_msec()-started<60000:await process_frame
	if not check(timeline.world!=null and timeline.loaded_index==midnight,day.id+" midnight timeline worker restoration"):return
	check(is_equal_approx(timeline.world.clock_seconds(),86400) and timeline.world.trains.size()==208,day.id+" midnight railway identity")
	check(timeline.clock_label.text.contains("00:00") and timeline.clock_label.text.contains("day 2"),day.id+" midnight clock identifies the next day")
	check(not timeline.services.is_empty(),day.id+" actual overnight traffic remains active")
	check(timeline.world.dispatcher().predictive.enabled==(day.dispatcher=="predictive"),day.id+" midnight dispatcher mode")
	var snapshot:=preload("res://sim/world_snapshot.gd")
	var state: Dictionary=snapshot.capture(timeline.world)
	var store:=preload("res://persistence/save_store.gd").new(OUT+"/midnight-"+day.id)
	var written:=store.write_slot("1",state,{title="Prepared midnight verification",timetable=day.id,day=2,clock="00:00:00"})
	if check(written.ok,day.id+" full 208-service midnight save written"):
		var read:=store.read_slot("1")
		if check(read.ok,day.id+" midnight save checksum verified"):
			var restored: Dictionary=snapshot.restore(read.data.checkpoint)
			if check(restored.ok,day.id+" midnight railway restored from save"):
				check(var_to_bytes(snapshot.capture(restored.world))==var_to_bytes(state),day.id+" midnight full railway survives save/load exactly")
	check(var_to_bytes(snapshot.capture(timeline.world))==var_to_bytes(state),day.id+" save checks leave the displayed railway unchanged")
	await shot(day.id+"-midnight")
